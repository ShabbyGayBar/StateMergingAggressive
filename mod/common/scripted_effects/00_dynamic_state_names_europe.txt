STATE_INGRIA_state_name_assign = {
	if = {
		limit = {
			is_template_used = rus_lenin_character_template
			owner = {
				country_has_primary_culture = cu:russian
				has_law_or_variant = law_type:law_council_republic
			}
		}
		reset_state_name = yes
        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_INGRIA_city_russian_sov # St. Petersburg only becomes Leningrad if Lenin has been around to be a namesake
        }
	}
	else_if = {
		limit = {
			owner = {
				country_has_primary_culture = cu:russian
				has_technology_researched = nationalism
				NOT = { has_law_or_variant = law_type:law_council_republic }

				OR = {
					any_rival_country = {
						any_primary_culture = {
							has_discrimination_trait = language_germanophone
						}
					}
					any_rivaling_country = {
						any_primary_culture = {
							has_discrimination_trait = language_germanophone
						}
					}
				}
			}
		}
		reset_state_name = yes
        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_INGRIA_city_russian_petrograd # St. Petersburg only becomes Petrograd when Germanophobia becomes a thing
        }
	}
	else_if = {
		limit = {
			owner = {
				country_has_primary_culture = cu:russian
			}
		}
		set_state_name = STATE_INGRIA_russian
        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_INGRIA_city_russian_imp
        }
	}
	else = { # switch back to defaults
		reset_state_name = yes
		reset_hub_names = yes
	}
}
