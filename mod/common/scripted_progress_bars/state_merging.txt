REPLACE:je_cuba_total_bar = {
 	name = "je_cuba_independencia_bar"
 	desc = {
 		first_valid = {
 			triggered_desc = {
 				trigger = {
 					s:STATE_MISSISSIPPI = {
 						any_scope_state = {
 							owner = { c:CSA ?= this }
 						}
 					}
 				}
 				desc = je_cuba_join_csa_bar
 			}
 			triggered_desc = {
 				trigger = {
 					s:STATE_MISSISSIPPI = {
 						any_scope_state = {
 							owner = { c:USA ?= this }
 						}
 					}
 				}
 				desc = je_cuba_join_usa_bar
 			}
 		}
 	}
 			
 	second_desc = "je_cuba_independencia_bar"

	progress_texture = "gfx/interface/progressbar/progressbar_default.dds"
	background_texture = "gfx/interface/progressbar/progressbar_gold.dds"
	marker = yes
	left_icon = "gfx/interface/progressbar/greatgame_bear_marker.dds"
	right_icon = "gfx/interface/progressbar/greatgame_lion_marker.dds"

	monthly_progress = {
		progress_condition = {
			limit = {
				OR = {
					c:USA ?= { has_or_is_contesting_state_in_state_region = STATE_MISSISSIPPI }
					c:CSA ?= { has_or_is_contesting_state_in_state_region = STATE_MISSISSIPPI }
				}
				OR = {
					exists = c:CUB
					AND = {
						exists = c:ATL
						c:ATL = { was_formed_from = CUB }
					}
				}
				owner = {
					has_journal_entry = je_cuba_independencia
					has_variable = independencia_target
					has_variable = independencia_current
				}
			}
			add = {
				desc = "je_cuba_total_drift_tt"
				value = owner.var:independencia_target
				subtract = owner.var:independencia_current
				multiply = je_cuba_drift_factor 
			}
			else = {
				add = {
					add = 1000
					desc = usa_does_not_exist_tt
				}
			}
		}
	}
	start_value = 35
	min_value = 0
	max_value = 100
}

				}
			}
			# Legitimacy below 75
			add = {
				desc = "lands_of_anarchy_legitimacy_tt"
				value = legitimacy
				subtract = 75
				divide = 30
			}

			add = {
				desc = "lands_of_anarchy_fez_education_access_tt"
				owner.capital ?= {
					value = literacy_rate
					multiply = 5
					subtract = 1
					max = 0.5
				}
			}

			add = {
				desc = "lands_of_anarchy_fez_infrastructure_tt"
				owner.capital ?= {
					value = infrastructure
					divide = 4.5
					subtract = 9
					max = 0.5
				}
			}
			# On Harka
			progress_condition = {
				limit = {
					has_modifier = lands_of_anarchy_harka_mod
				}
				add = {
					desc = "lands_of_anarchy_harka_popularity_mod"
					value = 1
				}
			}
			# lands_of_anarchy.4 - appointed IG tribe leader
			progress_condition = {
				limit = {
					owner = {
						any_scope_character = {
							has_modifier = lands_of_anarchy_tribal_marrakech_leader_mod
						}
					}
				}
				add = {
					desc = "lands_of_anarchy_tribe_leader_tt"
					value = 1.5
				}
			}
			# lands_of_anarchy.4 - rejected IG tribe leader
			progress_condition = {
				limit = {
					has_variable = tribal_leader_appointment_rejected
				}
				subtract = {
					desc = "lands_of_anarchy_tribe_leader_rejected_tt"
					value = 0.3
				}
			}
			# lands_of_anarchy.5 - sent medicine
			progress_condition = {

						has_modifier = lands_of_anarchy_tribe_received_medicine_mod
					}
				}
				add = {
					desc = "lands_of_anarchy_sent_medicine_to_tribes_tt"
					value = 1
				}
			}
			# lands_of_anarchy.5 - refused to send medicine
			progress_condition = {
				limit = {

						has_modifier = lands_of_anarchy_refused_medicine_mod

			}
			# The state is not ours anymore
